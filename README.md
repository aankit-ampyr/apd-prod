# clear the build
```bash
rm package-lock.json
rm -rf node_modules

rm packages/react-common/package-lock.json
rm -rf packages/react-common/node_modules

rm apps/*/package-lock.json
rm -rf apps/*/node_modules

rm package-lock.json; rm -rf node_modules; rm packages/react-common/package-lock.json; rm -rf packages/react-common/node_modules; rm apps/*/package-lock.json; rm -rf apps/*/node_modules
```
